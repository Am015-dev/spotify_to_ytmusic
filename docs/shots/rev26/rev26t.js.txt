// review rev26 (phonedrift-r25i): BRAKE+steer through the REAL touch buttons on the BUILT page, phone profile (iPhone UA, 852x393 touch, DPR 3, 6x CPU).
// node rev26t.js <base url> <city> [shot.jpg]
// Presses are trusted CDP touch events at the centres of #tB (BRAKE), #tG (GAS) and the right half of #btnZone (#tR), so the game's own
// touch handlers set TOUCH.*; the sim is stepped 1/60 in the page between presses (as the GOs step it). Street spawn as in GO feel/drive.
// Cases: bd50 = BRAKE+steer from 50 km/h (no gas) 2 s; bk50 = BRAKE alone 2 s; tapNNN = cornering at 40 km/h with GAS + right held,
// a BRAKE tap of NNN ms, then 1 s more: drift frames (RO.dDir) and max slip. A tap that drifts = an unwanted drift.
const {chromium}=require('/opt/node22/lib/node_modules/playwright');
const [BASE,C,SHOT]=process.argv.slice(2);
(async()=>{const b=await chromium.launch({args:['--use-angle=swiftshader','--enable-unsafe-swiftshader']});

 const pos=await EV(()=>{const c=s=>{const r=document.querySelector(s).getBoundingClientRect();return[r.left+r.width/2,r.top+r.height/2,r.width]};return{B:c('#tB'),G:c('#tG'),R:c('#tR')}});
 // CDP touch: send start/end per finger so each element sees its own touchstart / touchend

  for(const k of ids)if(!down.includes(k)){down=[...down,k];await cdp.send('Input.dispatchTouchEvent',{type:'touchStart',touchPoints:down.map(x=>({x:pos[x][0],y:pos[x][1],id:{R:1,G:2,B:3}[x]}))})}};

  /* settle: 1 s standing with no finger down, so the steer ramp, D26 hold timer and B2K latches of the previous case are cleared */RO.frozen=true;for(let i=0;i<60;i++){RO.v=0;roamStep(1/60)}roamWarp(q[0],q[1],q[2],true);
  RO.v=A.k/3.6;RO.vh=RO.h;RO.yr=0;RO.dl=0;RO.dDir=0;RO.dT=0;RO.dEx=0;RO.inv=99;RO.hp=100;RO.turbo=0;RO.stkT=0;RO.crTurn=null;camSnap=true;RO.frozen=true/* the page loop stops stepping; only this script steps */;window.__r26={h0:RO.vh,dr:0,en:-1,t90:-1,slip:0,f:0};return [Math.round(RO.v*3.6),TOUCH.dir,TOUCH.brake,TOUCH.gas]},{k:kmh});
 const step=n=>EV(A=>{const S=__r26,W=a=>Math.atan2(Math.sin(a),Math.cos(a));for(let i=0;i<A.n;i++){roamStep(1/60);try{roamCam(1/60)}catch(e){}S.f++;if(RO.dDir){S.dr++;if(S.en<0)S.en=+(S.f/60).toFixed(2)}
   S.slip=Math.max(S.slip,Math.abs(W(RO.h-RO.vh))*57.3);if(A.tr&&S.f>=28&&S.f<=50)(S.tr=S.tr||[]).push(S.f+':'+(TOUCH.brake?'B':'-')+(RO.dDir||0)+'/'+Math.round(Math.abs(W(RO.h-RO.vh))*57.3)+'/'+Math.round(RO.v*3.6)+'/'+(TOUCH.steer||0).toFixed(1));if(S.t90<0&&Math.abs(W(RO.vh-S.h0))>=Math.PI/2)S.t90=+(S.f/60).toFixed(2)}

 out.errs=errs.slice(0,3);console.log(JSON.stringify(out));await b.close()})().catch(e=>{console.log('ERR '+String(e).slice(0,300));process.exit(1)});
